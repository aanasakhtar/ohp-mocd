% ALGORITHM 1: OHP-MOCD Algorithmic Pseudocode
% --------------------------------------------------------------------------
\begin{algorithm}[t]
\caption{OHP-MOCD: Overlapping Hypergraph Pareto MOEA}\label{alg:ohpmocd}
\begin{algorithmic}[1]
\Require Network $G = (V, E)$, Population size $N_{\text{pop}}$, Generations $T_{\text{max}}$, Mutation rate $p_m$, Crossover rate $p_c$
\Ensure Non-dominated set of overlapping community covers $\mathcal{P}^*$
\State Construct Dual-Weighted Interaction (DWI) hypergraph $\mathcal{H} = (V, \mathcal{E}, W_e)$
\State Initialize population $\mathcal{P}_0 = \{\mathbf{x}_1, \dots, \mathbf{x}_{N_{\text{pop}}}\}$ via Boundary-Seeded Random Walk
\For{$t = 1$ to $T_{\text{max}}$}
    \State $\mathcal{Q}_t \gets \emptyset$
    \While{$|\mathcal{Q}_t| < N_{\text{pop}}$}
        \State Select parents $\mathbf{x}_a, \mathbf{x}_b \in \mathcal{P}_{t-1}$ via Binary Tournament
        \State $\mathbf{y} \gets \text{Hyperedge-Preserving Crossover}(\mathbf{x}_a, \mathbf{x}_b, p_c)$
        \State $\mathbf{y}' \gets \text{Boundary-Directed Mutation}(\mathbf{y}, p_m)$
        \State $\mathcal{Q}_t \gets \mathcal{Q}_t \cup \{\mathbf{y}'\}$
    \EndWhile
    \State Evaluate dual objectives $\mathbf{f}(\mathbf{x}) = (f_1(\mathbf{x}), f_2(\mathbf{x}))$ for all $\mathbf{x} \in \mathcal{P}_{t-1} \cup \mathcal{Q}_t$
    \State $\mathcal{P}_t \gets \text{Non-Dominated Sorting and Crowding Distance Selection}(\mathcal{P}_{t-1} \cup \mathcal{Q}_t, N_{\text{pop}})$
\EndFor
\State Extract Pareto Front $\mathcal{P}^* \gets \text{Front}_1(\mathcal{P}_{T_{\text{max}}})$
\State \Return Optimal partition $\mathbf{x}^* = \arg\max_{\mathbf{x} \in \mathcal{P}^*} Q_{\text{ov}}(\mathbf{x})$
\end{algorithmic}
\end{algorithm}

% THEOREM 1: Statistical Null Error Overlap Boundary Bound
% --------------------------------------------------------------------------
\begin{theorem}[Statistical Null Error Overlap Boundary Bound]\label{thm:overlap_bound}
Let $u \in V$ be a vertex with degree $d_u$ affiliated with $K_u \ge 2$ adjacent communities. Under the null hypothesis of uniform random edge distribution across candidate communities, the threshold $\theta_u$ separating true overlapping multi-membership from stochastic degree noise is bounded by:
\begin{equation}
\theta_u = \frac{1}{K_u} + \frac{1}{\sqrt{d_u}}
\end{equation}
A vertex $u$ is assigned to community $C_k$ if and only if its normalized support satisfies $\mathcal{S}(u, C_k) \ge \theta_u$.
\end{theorem}
\begin{proof}
Under the null configuration model where the $d_u$ incident edges of node $u$ are randomly distributed among $K_u$ clusters, the empirical membership proportion $\hat{p}_k = \frac{d_u(C_k)}{d_u}$ follows a multinomial distribution with mean $\mu = \frac{1}{K_u}$ and variance $\sigma^2 = \frac{1}{d_u} \frac{1}{K_u}\left(1 - \frac{1}{K_u}\right) \le \frac{1}{4 d_u}$. By the Central Limit Theorem and Chebyshev's inequality, any observed support exceeding $\frac{1}{K_u} + \frac{1}{\sqrt{d_u}}$ deviates from the null hypothesis by at least two standard errors ($p < 0.05$), rejecting accidental overlap.
\end{proof}
